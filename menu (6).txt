#!/bin/bash

###############################################################################
# Auto-Install to PATH
###############################################################################

# Check if script is in PATH
SCRIPT_NAME="menu"
SCRIPT_PATH="$(readlink -f "$0")"
INSTALL_PATH="/usr/local/bin/$SCRIPT_NAME"

# If not running from /usr/local/bin, offer to install
if [[ "$SCRIPT_PATH" != "$INSTALL_PATH" ]] && [[ ! -L "$INSTALL_PATH" ]]; then
    # Check if running as root
    if [[ $EUID -eq 0 ]]; then
        echo "[i] Installing 'menu' command to system PATH..."
        cp "$SCRIPT_PATH" "$INSTALL_PATH"
        chmod +x "$INSTALL_PATH"
        echo "[✓] Installation complete!"
        echo "[i] You can now type 'menu' from anywhere!"
        echo ""
        sleep 2
        # Re-execute from installed location
        exec "$INSTALL_PATH" "$@"
    else
        echo "[!] First-time setup: Installing 'menu' command..."
        echo "[i] This requires root privileges"
        echo ""
        if command -v sudo &> /dev/null; then
            sudo cp "$SCRIPT_PATH" "$INSTALL_PATH"
            sudo chmod +x "$INSTALL_PATH"
            echo "[✓] Installation complete!"
            echo "[i] You can now type 'menu' from anywhere!"
            echo ""
            sleep 2
            # Re-execute from installed location with sudo
            exec sudo "$INSTALL_PATH" "$@"
        else
            echo "[!] sudo not available. Please run as root:"
            echo "    su -"
            echo "    bash $SCRIPT_PATH"
            exit 1
        fi
    fi
fi

# If not root, re-execute with sudo
if [[ $EUID -ne 0 ]]; then
    if command -v sudo &> /dev/null; then
        exec sudo "$0" "$@"
    else
        echo "[!] This script requires root privileges"
        echo "[!] Please run: su - or install sudo"
        exit 1
    fi
fi

check_os() {
    if [[ -f /etc/os-release ]]; then
        . /etc/os-release
        OS=$ID
        VER=$VERSION_ID

        case "$OS" in
            debian|ubuntu|linuxmint|pop|elementary|zorin|kali|parrot|raspbian)
                print_success "Detected $PRETTY_NAME (Debian-family)"
                PKG_MANAGER="apt"
                ;;
            centos|rhel|rocky|alma|fedora|oracle)
                print_success "Detected $PRETTY_NAME (RHEL-family)"
                PKG_MANAGER="yum"
                if command -v dnf &> /dev/null; then
                    PKG_MANAGER="dnf"
                fi
                ;;
            arch|manjaro|endeavouros)
                print_success "Detected $PRETTY_NAME (Arch-family)"
                PKG_MANAGER="pacman"
                ;;
            opensuse*|sles)
                print_success "Detected $PRETTY_NAME (SUSE-family)"
                PKG_MANAGER="zypper"
                ;;
            alpine)
                print_success "Detected $PRETTY_NAME (Alpine)"
                PKG_MANAGER="apk"
                ;;
            *)
                print_warning "Detected $PRETTY_NAME - not officially tested"
                print_info "Attempting Debian-compatible installation..."
                PKG_MANAGER="apt"
                ;;
        esac
        return 0
    else
        print_error "Cannot detect OS"
        exit 1
    fi
}

check_internet() {
    print_info "Checking internet connectivity (IPv4 + IPv6)..."
    local ipv4_ok=false
    local ipv6_ok=false

    # Check IPv4 connectivity
    if ping -4 -c 1 -W 5 8.8.8.8 &> /dev/null; then
        ipv4_ok=true
        print_success "IPv4 connectivity: OK"
    else
        print_warning "IPv4 connectivity: FAILED"
    fi

    # Check IPv6 connectivity
    if ping -6 -c 1 -W 5 2001:4860:4860::8888 &> /dev/null; then
        ipv6_ok=true
        print_success "IPv6 connectivity: OK"
    else
        print_warning "IPv6 connectivity: Not available (IPv4-only mode)"
    fi

    if [[ "$ipv4_ok" == false ]] && [[ "$ipv6_ok" == false ]]; then
        print_error "No internet connection (neither IPv4 nor IPv6)"
        exit 1
    fi

    # Store global connectivity flags
    HAS_IPV4=$ipv4_ok
    HAS_IPV6=$ipv6_ok
    return 0
}

###############################################################################
# UNIVERSAL DEPENDENCY INSTALLER (Fixes Debian 13 / any OS stuck issues)
# - Uses timeout to prevent hangs
# - Kills stale dpkg/apt locks
# - Handles all package managers
# - Non-interactive forced mode
###############################################################################

fix_dpkg_locks() {
    # Kill any stuck apt/dpkg processes
    pkill -9 -f "apt" 2>/dev/null
    pkill -9 -f "dpkg" 2>/dev/null
    sleep 1

    # Remove stale lock files
    rm -f /var/lib/dpkg/lock-frontend 2>/dev/null
    rm -f /var/lib/dpkg/lock 2>/dev/null
    rm -f /var/cache/apt/archives/lock 2>/dev/null
    rm -f /var/lib/apt/lists/lock 2>/dev/null

    # Reconfigure dpkg if interrupted
    dpkg --configure -a --force-confdef --force-confold 2>/dev/null
}

install_packages() {
    local packages=("$@")
    local max_retries=3
    local retry=0

    # Set fully non-interactive environment
    export DEBIAN_FRONTEND=noninteractive
    export NEEDRESTART_MODE=a
    export NEEDRESTART_SUSPEND=1
    export UCF_FORCE_CONFFOLD=1

    while [[ $retry -lt $max_retries ]]; do
        case "$PKG_MANAGER" in
            apt)
                # Fix any stuck locks first
                fix_dpkg_locks

                # Configure apt for non-interactive, no prompts
                local APT_OPTS="-o Dpkg::Options::=--force-confdef -o Dpkg::Options::=--force-confold -o APT::Get::Allow-Unauthenticated=true"

                # Update with timeout (Debian 13 can hang here)
                timeout 120 apt-get update $APT_OPTS -qq 2>/dev/null

                # Install with timeout and force-yes
                if timeout 300 apt-get install -y -qq $APT_OPTS --no-install-recommends "${packages[@]}" 2>/dev/null; then
                    return 0
                fi

                # If failed, try fixing and retrying
                fix_dpkg_locks
                apt-get -f install -y -qq $APT_OPTS 2>/dev/null
                ;;
            dnf)
                if timeout 300 dnf install -y -q "${packages[@]}" 2>/dev/null; then
                    return 0
                fi
                ;;
            yum)
                if timeout 300 yum install -y -q "${packages[@]}" 2>/dev/null; then
                    return 0
                fi
                ;;
            pacman)
                if timeout 300 pacman -S --noconfirm --needed "${packages[@]}" 2>/dev/null; then
                    return 0
                fi
                ;;
            zypper)
                if timeout 300 zypper install -y -n "${packages[@]}" 2>/dev/null; then
                    return 0
                fi
                ;;
            apk)
                if timeout 300 apk add --no-cache "${packages[@]}" 2>/dev/null; then
                    return 0
                fi
                ;;
        esac

        ((retry++))
        print_warning "Package install attempt $retry/$max_retries failed, retrying..."
        sleep 3
    done

    print_error "Failed to install packages after $max_retries attempts: ${packages[*]}"
    return 1
}

update_package_cache() {
    export DEBIAN_FRONTEND=noninteractive
    export NEEDRESTART_MODE=a
    export NEEDRESTART_SUSPEND=1

    case "$PKG_MANAGER" in
        apt)
            fix_dpkg_locks
            timeout 120 apt-get update -qq \
                -o Dpkg::Options::=--force-confdef \
                -o Dpkg::Options::=--force-confold 2>/dev/null
            ;;
        dnf) timeout 60 dnf makecache -q 2>/dev/null ;;
        yum) timeout 60 yum makecache -q 2>/dev/null ;;
        pacman) timeout 60 pacman -Sy --noconfirm 2>/dev/null ;;
        zypper) timeout 60 zypper refresh -q 2>/dev/null ;;
        apk) timeout 60 apk update -q 2>/dev/null ;;
    esac
}

install_base_dependencies() {
    print_info "Installing base dependencies (universal installer)..."

    update_package_cache

    # Core packages needed by all protocols
    local core_pkgs=(curl wget jq openssl ca-certificates)

    # Additional packages
    local extra_pkgs=()
    case "$PKG_MANAGER" in
        apt)
            extra_pkgs=(unzip xz-utils qrencode ethtool net-tools procps bc socat cron)
            # iptables-persistent needs special handling on Debian 13+
            echo iptables-persistent iptables-persistent/autosave_v4 boolean true | debconf-set-selections 2>/dev/null
            echo iptables-persistent iptables-persistent/autosave_v6 boolean true | debconf-set-selections 2>/dev/null
            extra_pkgs+=(iptables iptables-persistent)
            ;;
        dnf|yum)
            extra_pkgs=(unzip xz qrencode ethtool net-tools procps-ng bc socat cronie iptables iptables-services)
            ;;
        pacman)
            extra_pkgs=(unzip xz qrencode ethtool net-tools procps-ng bc socat cronie iptables)
            ;;
        zypper)
            extra_pkgs=(unzip xz qrencode ethtool net-tools procps bc socat cron iptables)
            ;;
        apk)
            extra_pkgs=(unzip xz qrencode ethtool net-tools procps bc socat iptables)
            ;;
    esac

    install_packages "${core_pkgs[@]}" "${extra_pkgs[@]}"

    # Ensure cron is running
    if command -v systemctl &> /dev/null; then
        systemctl enable cron 2>/dev/null || systemctl enable crond 2>/dev/null || true
        systemctl start cron 2>/dev/null || systemctl start crond 2>/dev/null || true
    fi

    print_success "Base dependencies installed"
}

###############################################################################
# IPv4/IPv6 Dual-Stack IP Detection Functions
###############################################################################

get_all_ips() {
    ip -4 addr show | grep -oP '(?<=inet\s)\d+(\.\d+){3}' | grep -v '127.0.0.1'
}

get_all_ipv6() {
    ip -6 addr show scope global | grep -oP '(?<=inet6\s)[0-9a-f:]+' | grep -v '^::1$'
}

get_main_ip() {
    local ip
    for svc in ifconfig.me icanhazip.com ipinfo.io/ip api.ipify.org; do
        ip=$(curl -4 -s --connect-timeout 5 --max-time 10 "$svc" 2>/dev/null)
        if [[ -n "$ip" ]] && [[ "$ip" =~ ^[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+$ ]]; then
            echo "$ip"
            return 0
        fi
    done
    ip=$(ip -4 addr show | grep -oP '(?<=inet\s)\d+(\.\d+){3}' | grep -v '127.0.0.1' | head -n 1)
    if [[ -n "$ip" ]]; then
        echo "$ip"
        return 0
    fi
    echo "127.0.0.1"
}

get_main_ipv6() {
    local ip6
    for svc in ifconfig.me icanhazip.com api64.ipify.org; do
        ip6=$(curl -6 -s --connect-timeout 5 --max-time 10 "$svc" 2>/dev/null)
        if [[ -n "$ip6" ]] && [[ "$ip6" =~ : ]]; then
            echo "$ip6"
            return 0
        fi
    done
    ip6=$(ip -6 addr show scope global | grep -oP '(?<=inet6\s)[0-9a-f:]+' | head -n 1)
    if [[ -n "$ip6" ]]; then
        echo "$ip6"
        return 0
    fi
    echo ""
}

is_ipv6() {
    [[ "$1" =~ : ]]
}

format_ip_for_url() {
    if is_ipv6 "$1"; then
        echo "[$1]"
    else
        echo "$1"
    fi
}

format_ip_suffix() {
    echo "$1" | tr '.:' '--'
}

detect_ip_stack() {
    local ipv4=$(get_main_ip)
    local ipv6=$(get_main_ipv6)

    echo ""
    echo -e "${CYAN}=== Server IP Stack Detection ===${NC}"
    if [[ -n "$ipv4" ]] && [[ "$ipv4" != "127.0.0.1" ]]; then
        echo -e "  ${GREEN}[✓]${NC} IPv4: $ipv4"
    else
        echo -e "  ${RED}[✗]${NC} IPv4: Not detected"
    fi

    if [[ -n "$ipv6" ]]; then
        echo -e "  ${GREEN}[✓]${NC} IPv6: $ipv6"
    else
        echo -e "  ${YELLOW}[!]${NC} IPv6: Not detected (IPv4-only server)"
    fi
    echo ""
}

setup_ipv6_firewall() {
    print_info "Configuring firewall rules for VPN services (universal)..."

    # Open essential ports using universal firewall opener
    firewall_open_port 443 tcp
    firewall_open_port 80 tcp

    # Also allow WebUI port if configured
    if [[ -f "$WEBUI_PORT_FILE" ]]; then
        local wport=$(cat "$WEBUI_PORT_FILE" 2>/dev/null)
        if [[ -n "$wport" ]]; then
            firewall_open_port "$wport" tcp
        fi
    fi

    # IPv6 base rules (loopback, ICMP, established) - use iptables if available
    if command -v ip6tables &> /dev/null; then
        ip6tables -C INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT 2>/dev/null || \
            ip6tables -A INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT 2>/dev/null
        ip6tables -C INPUT -p ipv6-icmp -j ACCEPT 2>/dev/null || \
            ip6tables -A INPUT -p ipv6-icmp -j ACCEPT 2>/dev/null
        ip6tables -C INPUT -i lo -j ACCEPT 2>/dev/null || \
            ip6tables -A INPUT -i lo -j ACCEPT 2>/dev/null
    fi

    # nftables base rules for IPv6 (Debian 13+)
    if command -v nft &> /dev/null; then
        if systemctl is-active --quiet nftables 2>/dev/null; then
            nft add rule inet filter input ct state established,related accept 2>/dev/null
            nft add rule inet filter input iif lo accept 2>/dev/null
            nft add rule inet filter input meta l4proto ipv6-icmp accept 2>/dev/null
        fi
    fi

    firewall_save
    print_success "Firewall configured (universal)"
}

setup_dual_stack_kernel() {
    print_info "Configuring kernel for dual-stack IPv4/IPv6..."

    sysctl -w net.ipv6.conf.all.disable_ipv6=0 > /dev/null 2>&1
    sysctl -w net.ipv6.conf.default.disable_ipv6=0 > /dev/null 2>&1
    sysctl -w net.ipv6.conf.lo.disable_ipv6=0 > /dev/null 2>&1
    sysctl -w net.ipv6.conf.all.forwarding=1 > /dev/null 2>&1
    sysctl -w net.ipv6.conf.default.forwarding=1 > /dev/null 2>&1
    sysctl -w net.ipv6.conf.all.accept_ra=2 > /dev/null 2>&1
    sysctl -w net.ipv6.conf.default.accept_ra=2 > /dev/null 2>&1
    sysctl -w net.ipv6.bindv6only=0 > /dev/null 2>&1

    # Persist
    for param in \
        "net.ipv6.conf.all.disable_ipv6=0" \
        "net.ipv6.conf.default.disable_ipv6=0" \
        "net.ipv6.conf.lo.disable_ipv6=0" \
        "net.ipv6.conf.all.forwarding=1" \
        "net.ipv6.conf.default.forwarding=1" \
        "net.ipv6.conf.all.accept_ra=2" \
        "net.ipv6.conf.default.accept_ra=2" \
        "net.ipv6.bindv6only=0"; do
        local key="${param%%=*}"
        sed -i "/^${key//./\\.}/d" /etc/sysctl.conf 2>/dev/null
        echo "$param" >> /etc/sysctl.conf
    done

    sysctl -p > /dev/null 2>&1
    print_success "Dual-stack kernel configured"
}

###############################################################################
# Universal Firewall Port Opener (Works on ANY OS / ANY Firewall)
# Auto-detects: ufw, firewalld, nftables, iptables
# Opens port on both IPv4 and IPv6 automatically
# Usage: firewall_open_port <port> [protocol] [ip_address]
#   port       - port number (required)
#   protocol   - tcp/udp/both (default: tcp)
#   ip_address - optional: restrict to specific IP (for additional IPs)
###############################################################################

firewall_open_port() {
    local port="$1"
    local proto="${2:-tcp}"
    local bind_ip="${3:-}"
    
    if [[ -z "$port" ]]; then
        print_error "firewall_open_port: port number required"
        return 1
    fi

    print_info "Opening port $port/$proto in firewall..."

    local opened=false

    # Method 1: ufw (Ubuntu/Debian default)
    if command -v ufw &> /dev/null; then
        local ufw_status=$(ufw status 2>/dev/null | head -1)
        if echo "$ufw_status" | grep -qi "active"; then
            if [[ "$proto" == "both" ]]; then
                ufw allow "$port" > /dev/null 2>&1
            else
                ufw allow "$port/$proto" > /dev/null 2>&1
            fi
            opened=true
            print_success "Port $port/$proto opened via ufw"
        fi
    fi

    # Method 2: firewalld (RHEL/CentOS/Fedora/openSUSE)
    if command -v firewall-cmd &> /dev/null; then
        local fwd_state=$(firewall-cmd --state 2>/dev/null)
        if [[ "$fwd_state" == "running" ]]; then
            if [[ "$proto" == "both" ]]; then
                firewall-cmd --permanent --add-port="$port/tcp" > /dev/null 2>&1
                firewall-cmd --permanent --add-port="$port/udp" > /dev/null 2>&1
            else
                firewall-cmd --permanent --add-port="$port/$proto" > /dev/null 2>&1
            fi
            firewall-cmd --reload > /dev/null 2>&1
            opened=true
            print_success "Port $port/$proto opened via firewalld"
        fi
    fi

    # Method 3: nftables (Debian 13+ default, modern distros)
    if command -v nft &> /dev/null; then
        # Check if nftables is the active firewall (not just installed alongside iptables)
        if systemctl is-active --quiet nftables 2>/dev/null || \
           [[ -f /etc/nftables.conf ]] && ! command -v iptables-legacy &> /dev/null; then
            
            # Ensure input chain exists in inet filter table
            nft list tables 2>/dev/null | grep -q "inet filter" || \
                nft add table inet filter 2>/dev/null
            nft list chain inet filter input 2>/dev/null || \
                nft add chain inet filter input '{ type filter hook input priority 0; policy accept; }' 2>/dev/null

            if [[ "$proto" == "both" ]]; then
                if [[ -n "$bind_ip" ]]; then
                    if is_ipv6 "$bind_ip" 2>/dev/null; then
                        nft add rule inet filter input ip6 daddr "$bind_ip" tcp dport "$port" accept 2>/dev/null
                        nft add rule inet filter input ip6 daddr "$bind_ip" udp dport "$port" accept 2>/dev/null
                    else
                        nft add rule inet filter input ip daddr "$bind_ip" tcp dport "$port" accept 2>/dev/null
                        nft add rule inet filter input ip daddr "$bind_ip" udp dport "$port" accept 2>/dev/null
                    fi
                else
                    nft add rule inet filter input tcp dport "$port" accept 2>/dev/null
                    nft add rule inet filter input udp dport "$port" accept 2>/dev/null
                fi
            else
                if [[ -n "$bind_ip" ]]; then
                    if is_ipv6 "$bind_ip" 2>/dev/null; then
                        nft add rule inet filter input ip6 daddr "$bind_ip" "$proto" dport "$port" accept 2>/dev/null
                    else
                        nft add rule inet filter input ip daddr "$bind_ip" "$proto" dport "$port" accept 2>/dev/null
                    fi
                else
                    nft add rule inet filter input "$proto" dport "$port" accept 2>/dev/null
                fi
            fi

            # Save nftables rules
            nft list ruleset > /etc/nftables.conf 2>/dev/null
            opened=true
            print_success "Port $port/$proto opened via nftables"
        fi
    fi

    # Method 4: iptables + ip6tables (legacy fallback, works everywhere)
    if command -v iptables &> /dev/null; then
        local do_iptables=true
        # Skip if ufw or firewalld already handled it (they use iptables backend)
        if [[ "$opened" == true ]]; then
            do_iptables=false
        fi

        if [[ "$do_iptables" == true ]]; then
            local protos=()
            if [[ "$proto" == "both" ]]; then
                protos=(tcp udp)
            else
                protos=("$proto")
            fi

            for p in "${protos[@]}"; do
                # IPv4
                if [[ -n "$bind_ip" ]] && ! is_ipv6 "$bind_ip" 2>/dev/null; then
                    iptables -C INPUT -d "$bind_ip" -p "$p" --dport "$port" -j ACCEPT 2>/dev/null || \
                        iptables -A INPUT -d "$bind_ip" -p "$p" --dport "$port" -j ACCEPT 2>/dev/null
                elif [[ -z "$bind_ip" ]]; then
                    iptables -C INPUT -p "$p" --dport "$port" -j ACCEPT 2>/dev/null || \
                        iptables -A INPUT -p "$p" --dport "$port" -j ACCEPT 2>/dev/null
                fi

                # IPv6
                if command -v ip6tables &> /dev/null; then
                    if [[ -n "$bind_ip" ]] && is_ipv6 "$bind_ip" 2>/dev/null; then
                        ip6tables -C INPUT -d "$bind_ip" -p "$p" --dport "$port" -j ACCEPT 2>/dev/null || \
                            ip6tables -A INPUT -d "$bind_ip" -p "$p" --dport "$port" -j ACCEPT 2>/dev/null
                    elif [[ -z "$bind_ip" ]]; then
                        ip6tables -C INPUT -p "$p" --dport "$port" -j ACCEPT 2>/dev/null || \
                            ip6tables -A INPUT -p "$p" --dport "$port" -j ACCEPT 2>/dev/null
                    fi
                fi
            done
            opened=true
            print_success "Port $port/$proto opened via iptables"
        fi
    fi

    # Save firewall rules persistently
    firewall_save

    if [[ "$opened" == false ]]; then
        print_warning "No supported firewall detected. Port $port may already be accessible."
        print_info "If connections fail, manually open port $port/$proto in your firewall/cloud panel."
    fi

    return 0
}

# Save firewall rules persistently across reboots
firewall_save() {
    # netfilter-persistent (Debian/Ubuntu with iptables-persistent)
    if command -v netfilter-persistent &> /dev/null; then
        netfilter-persistent save 2>/dev/null
        return 0
    fi

    # iptables-save direct (fallback for Debian/Ubuntu)
    if command -v iptables-save &> /dev/null; then
        mkdir -p /etc/iptables 2>/dev/null
        iptables-save > /etc/iptables/rules.v4 2>/dev/null
        if command -v ip6tables-save &> /dev/null; then
            ip6tables-save > /etc/iptables/rules.v6 2>/dev/null
        fi
    fi

    # firewalld (auto-saves with --permanent flag)
    # nftables (already saved to /etc/nftables.conf above)

    # RHEL/CentOS iptables-services
    if command -v service &> /dev/null && [[ -f /etc/sysconfig/iptables ]]; then
        service iptables save 2>/dev/null
        service ip6tables save 2>/dev/null
    fi

    return 0
}

# Open multiple ports at once
# Usage: firewall_open_ports "80 443 2053" [protocol]
firewall_open_ports() {
    local ports="$1"
    local proto="${2:-tcp}"
    for port in $ports; do
        firewall_open_port "$port" "$proto"
    done
}

enable_bbr() {
    print_info "Enabling BBR congestion control..."

    if ! lsmod | grep -q tcp_bbr; then
        modprobe tcp_bbr 2>/dev/null
        if ! grep -q "tcp_bbr" /etc/modules-load.d/bbr.conf 2>/dev/null; then
            echo "tcp_bbr" > /etc/modules-load.d/bbr.conf
        fi
    fi

    sysctl -w net.core.default_qdisc=fq > /dev/null 2>&1
    sysctl -w net.ipv4.tcp_congestion_control=bbr > /dev/null 2>&1

    if sysctl net.ipv4.tcp_congestion_control 2>/dev/null | grep -q bbr; then
        print_success "BBR congestion control enabled"
    else
        print_warning "BBR not available, using default congestion control"
    fi
}

optimize_network() {

    # Remove old entries to avoid duplicates
    local params=(

    for p in "${params[@]}"; do
        sed -i "/^${p//./\\.}/d" /etc/sysctl.conf 2>/dev/null
    done

net.ipv4.udp_wmem_min=8192

net.core.somaxconn=65535

net.ipv4.tcp_fin_timeout=10
net.ipv4.tcp_max_tw_buckets=2000000

# Aggressive keepalive (faster dead connection detection)

net.ipv4.tcp_keepalive_intvl=10
net.ipv4.tcp_keepalive_probes=6

# MTU probing (auto-discover optimal MTU, critical for IPv6)
net.ipv4.tcp_mtu_probing=1

# TCP timestamps, SACK, window scaling
net.ipv4.tcp_timestamps=1
net.ipv4.tcp_sack=1
net.ipv4.tcp_window_scaling=1

# Widen ephemeral port range

# Handle orphaned connections aggressively

net.ipv4.tcp_abort_on_overflow=0

    print_success "Advanced network optimizations applied"
}

optimize_mtu() {
    print_info "Detecting and optimizing MTU for all interfaces..."

    local iface=$(ip route show default | awk '/default/ {print $5}' | head -n 1)
    if [[ -z "$iface" ]]; then
        iface=$(ip -6 route show default | awk '/default/ {print $5}' | head -n 1)
    fi

    if [[ -n "$iface" ]]; then
        local current_mtu=$(ip link show "$iface" | awk '/mtu/ {print $5}')
        print_info "Interface: $iface, Current MTU: $current_mtu"

        if [[ "$current_mtu" -lt 1280 ]]; then
            ip link set "$iface" mtu 1280
            print_warning "MTU was below IPv6 minimum (1280), adjusted"
        fi

        # MSS clamping for better compatibility
        if command -v iptables &> /dev/null; then
            iptables -t mangle -C FORWARD -p tcp --tcp-flags SYN,RST SYN -j TCPMSS --clamp-mss-to-pmtu 2>/dev/null || \
                iptables -t mangle -A FORWARD -p tcp --tcp-flags SYN,RST SYN -j TCPMSS --clamp-mss-to-pmtu 2>/dev/null
        fi
        if command -v ip6tables &> /dev/null; then
            ip6tables -t mangle -C FORWARD -p tcp --tcp-flags SYN,RST SYN -j TCPMSS --clamp-mss-to-pmtu 2>/dev/null || \
                ip6tables -t mangle -A FORWARD -p tcp --tcp-flags SYN,RST SYN -j TCPMSS --clamp-mss-to-pmtu 2>/dev/null
        fi

        print_success "MTU optimization applied (MSS clamping enabled)"
    else
        print_warning "Could not detect default network interface"
    fi
}

setup_tcp_tuned_profile() {
    print_info "Setting up optimized TCP profile..."

    local iface=$(ip route show default | awk '/default/ {print $5}' | head -n 1)
    if [[ -n "$iface" ]]; then
        ethtool -K "$iface" gro on 2>/dev/null
        ethtool -K "$iface" gso on 2>/dev/null
        ethtool -K "$iface" tso on 2>/dev/null
        ethtool -K "$iface" tx-checksum-ip-generic on 2>/dev/null
        ethtool -K "$iface" rx-checksum on 2>/dev/null

        # Increase ring buffer if possible
        ethtool -G "$iface" rx 4096 tx 4096 2>/dev/null

        print_success "NIC offloading optimized on $iface"
    fi
}

###############################################################################

add_trojan_on_ip() {
    clear
    echo -e "${GREEN}╔════════════════════════════════════════════════╗${NC}"
    echo -e "${GREEN}║   Add Trojan on Additional IP (Dual-Stack)    ║${NC}"
    echo -e "${GREEN}╚════════════════════════════════════════════════╝${NC}"
    echo ""

    if ! systemctl is-active --quiet trojan; then
        print_error "Main Trojan service is not installed"
        print_info "Please install Trojan first (option 1)"
        read -p "Press Enter to continue..."
        return 1
    fi

    print_info "Detecting server IP addresses (IPv4 + IPv6)..."
    echo ""
    mapfile -t IPV4S < <(get_all_ips)
    mapfile -t IPV6S < <(get_all_ipv6)

    local ALL_IPS=()
    local idx=1

    echo -e "${CYAN}Available IPv4 addresses:${NC}"
    for ip in "${IPV4S[@]}"; do
        echo "  $idx. $ip"
        ALL_IPS+=("$ip")
        ((idx++))
    done

    if [[ ${#IPV6S[@]} -gt 0 ]]; then
        echo ""
        echo -e "${CYAN}Available IPv6 addresses:${NC}"
        for ip in "${IPV6S[@]}"; do
            echo "  $idx. $ip"
            ALL_IPS+=("$ip")
            ((idx++))
        done
    fi

    echo "  $idx. Enter custom IP (IPv4 or IPv6)"
    echo ""

    print_prompt "Select IP address (1-$idx):"
    read -r IP_CHOICE

    if [[ $IP_CHOICE -eq $idx ]]; then
        print_prompt "Enter IP address (IPv4 or IPv6):"
        read -r ADDITIONAL_IP
    elif [[ $IP_CHOICE -ge 1 ]] && [[ $IP_CHOICE -lt $idx ]]; then
        ADDITIONAL_IP="${ALL_IPS[$((IP_CHOICE-1))]}"
    else
        print_error "Invalid choice"
        read -p "Press Enter to continue..."
        return 1
    fi

    print_info "Using IP: $ADDITIONAL_IP"

    IP_SUFFIX=$(format_ip_suffix "$ADDITIONAL_IP")
    SERVICE_NAME="trojan-$IP_SUFFIX"
    CONFIG_FILE="/usr/local/etc/trojan/config-$IP_SUFFIX.json"

    if systemctl is-active --quiet "$SERVICE_NAME"; then
        print_warning "Trojan is already running on $ADDITIONAL_IP"
        read -p "Press Enter to continue..."
        return 0
    fi

    NEW_PASSWORD=$(generate_random_password)

    if [[ -f "/usr/local/etc/trojan/config.json" ]]; then
        cp /usr/local/etc/trojan/config.json "$CONFIG_FILE"
        jq --arg ip "$ADDITIONAL_IP" '.local_addr = $ip' "$CONFIG_FILE" > "$CONFIG_FILE.tmp" && mv "$CONFIG_FILE.tmp" "$CONFIG_FILE"
        jq --arg pass "$NEW_PASSWORD" '.password = [$pass]' "$CONFIG_FILE" > "$CONFIG_FILE.tmp" && mv "$CONFIG_FILE.tmp" "$CONFIG_FILE"
        print_success "Configuration created"
    else
        print_error "Base configuration not found"
        return 1
    fi

    # Open port 443 for this additional IP (universal firewall)
    firewall_open_port 443 tcp "$ADDITIONAL_IP"

    cat > "/etc/systemd/system/$SERVICE_NAME.service" << EOF
[Unit]
Description=Trojan-GFW Server on $ADDITIONAL_IP (Dual-Stack v4.0)
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
ExecStart=/usr/local/bin/trojan -c $CONFIG_FILE

            [[ -z "$XRAY_VER" || "$XRAY_VER" == "null" ]] && XRAY_VER="v25.3.6"
            local ARCH=$(uname -m)
            case "$ARCH" in
                x86_64|amd64) XRAY_ARCH="64" ;;
                aarch64|arm64) XRAY_ARCH="arm64-v8a" ;;
                *) XRAY_ARCH="64" ;;

###############################################################################
# Speed Test Module (IPv4 vs IPv6)
###############################################################################

speed_test() {
    clear
    echo -e "${GREEN}╔════════════════════════════════════════════════╗${NC}"
    echo -e "${GREEN}║   Network Speed & Latency Test                ║${NC}"
    echo -e "${GREEN}╚════════════════════════════════════════════════╝${NC}"
    echo ""

    # IPv4 latency
    echo -e "${CYAN}=== IPv4 Latency Test ===${NC}"
    for target in "8.8.8.8:Google DNS" "1.1.1.1:Cloudflare"; do
        local ip="${target%%:*}"
        local name="${target##*:}"
        if ping -4 -c 5 -W 3 "$ip" &> /dev/null; then
            local ms=$(ping -4 -c 5 -W 3 "$ip" 2>/dev/null | tail -1 | awk -F'/' '{print $5}')
            echo -e "  $name ($ip): ${GREEN}${ms}ms${NC}"
        else
            echo -e "  $name ($ip): ${RED}Unreachable${NC}"
        fi
    done
    echo ""

    # IPv6 latency
    echo -e "${CYAN}=== IPv6 Latency Test ===${NC}"
    for target in "2001:4860:4860::8888:Google DNS" "2606:4700:4700::1111:Cloudflare"; do
        local ip="${target%%:*:*:*:*}"
        local name="${target##*:}"
        # Re-extract properly
        ip=$(echo "$target" | rev | cut -d: -f2- | rev)
        name=$(echo "$target" | rev | cut -d: -f1 | rev)
        if ping -6 -c 5 -W 3 "$ip" &> /dev/null; then
            local ms=$(ping -6 -c 5 -W 3 "$ip" 2>/dev/null | tail -1 | awk -F'/' '{print $5}')
            echo -e "  $name: ${GREEN}${ms}ms${NC}"
        else
            echo -e "  $name: ${YELLOW}Not available${NC}"
        fi
    done
    echo ""

    # Download speed
    echo -e "${CYAN}=== Download Speed Test ===${NC}"
    print_info "Testing IPv4 download speed..."
    local ipv4_speed=$(curl -4 -s -o /dev/null -w '%{speed_download}' --connect-timeout 10 --max-time 15 http://speedtest.tele2.net/1MB.zip 2>/dev/null)
    if [[ -n "$ipv4_speed" ]] && [[ "$ipv4_speed" != "0.000" ]]; then
        local ipv4_mbps=$(echo "scale=2; $ipv4_speed / 1048576" | bc 2>/dev/null)
        echo -e "  IPv4 Download: ${GREEN}${ipv4_mbps} MB/s${NC}"
    else
        echo -e "  IPv4 Download: ${RED}Failed${NC}"
    fi

    print_info "Testing IPv6 download speed..."
    local ipv6_speed=$(curl -6 -s -o /dev/null -w '%{speed_download}' --connect-timeout 10 --max-time 15 http://speedtest.tele2.net/1MB.zip 2>/dev/null)
    if [[ -n "$ipv6_speed" ]] && [[ "$ipv6_speed" != "0.000" ]]; then
        local ipv6_mbps=$(echo "scale=2; $ipv6_speed / 1048576" | bc 2>/dev/null)
        echo -e "  IPv6 Download: ${GREEN}${ipv6_mbps} MB/s${NC}"
    else
        echo -e "  IPv6 Download: ${YELLOW}Not available${NC}"
    fi
    echo ""

    # Optimization status
    echo -e "${CYAN}=== Optimization Status ===${NC}"
    local cc=$(sysctl -n net.ipv4.tcp_congestion_control 2>/dev/null)
    echo -e "  Congestion Control: ${CYAN}$cc${NC} $([ "$cc" == "bbr" ] && echo -e "${GREEN}(optimal)${NC}")"

    local tfo=$(sysctl -n net.ipv4.tcp_fastopen 2>/dev/null)
    echo -e "  TCP Fast Open: ${CYAN}$tfo${NC} $([ "$tfo" == "3" ] && echo -e "${GREEN}(client+server)${NC}")"

    local qdisc=$(sysctl -n net.core.default_qdisc 2>/dev/null)
    echo -e "  Queue Discipline: ${CYAN}$qdisc${NC}"

    local nofile=$(ulimit -n 2>/dev/null)
    echo -e "  File Descriptors: ${CYAN}$nofile${NC}"

    local bv6=$(sysctl -n net.ipv6.bindv6only 2>/dev/null)
    echo -e "  Dual-Stack Binding: $([ "$bv6" == "0" ] && echo -e "${GREEN}Enabled${NC}" || echo -e "${RED}Disabled${NC}")"

    echo ""
    read -p "Press Enter to continue..."
}

repair_dual_stack() {
    clear
    echo -e "${GREEN}╔════════════════════════════════════════════════╗${NC}"
    echo -e "${GREEN}║   Repair Dual-Stack IPv4/IPv6 (v4.0)          ║${NC}"
    echo -e "${GREEN}╚════════════════════════════════════════════════╝${NC}"
    echo ""

    print_info "Running comprehensive dual-stack repair..."
    echo ""

    echo -e "${CYAN}Step 1: Fixing kernel IPv6 settings...${NC}"
    setup_dual_stack_kernel
    echo ""

    echo -e "${CYAN}Step 2: Fixing IPv6 firewall rules...${NC}"
    setup_ipv6_firewall
    echo ""

    # Fix Trojan
    if [[ -f /usr/local/etc/trojan/config.json ]]; then
        echo -e "${CYAN}Step 3: Fixing Trojan configuration...${NC}"
        local current_addr=$(jq -r '.local_addr' /usr/local/etc/trojan/config.json 2>/dev/null)
        if [[ "$current_addr" != "::" ]]; then
            jq '.local_addr = "::"' /usr/local/etc/trojan/config.json > /tmp/trojan-fix.json && \
                mv /tmp/trojan-fix.json /usr/local/etc/trojan/config.json
            print_success "Trojan config fixed to listen on :: (dual-stack)"
        else
            print_success "Trojan already configured for dual-stack"
        fi

        jq '.tcp.keep_alive = true | .tcp.no_delay = true | .tcp.reuse_port = true | .tcp.fast_open = true | .tcp.prefer_ipv4 = false' \
            /usr/local/etc/trojan/config.json > /tmp/trojan-fix.json && \
            mv /tmp/trojan-fix.json /usr/local/etc/trojan/config.json 2>/dev/null

        if systemctl is-active --quiet trojan; then
            systemctl restart trojan
            sleep 2
            systemctl is-active --quiet trojan && print_success "Trojan restarted" || print_error "Trojan failed to restart"
        fi
        echo ""
    fi

    # Fix Xray
    if [[ -f /usr/local/etc/xray/config.json ]]; then
        echo -e "${CYAN}Step 4: Fixing Xray configuration...${NC}"
        local xray_listen=$(jq -r '.inbounds[0].listen' /usr/local/etc/xray/config.json 2>/dev/null)
        if [[ "$xray_listen" != "::" ]]; then
            jq '.inbounds[0].listen = "::"' /usr/local/etc/xray/config.json > /tmp/xray-fix.json && \
                mv /tmp/xray-fix.json /usr/local/etc/xray/config.json
            print_success "Xray config fixed to listen on :: (dual-stack)"
        else
            print_success "Xray already configured for dual-stack"
        fi

        # Fix sockopt
        jq '.inbounds[0].streamSettings.sockopt = {"tcpKeepAliveIdle": 60, "tcpKeepAliveInterval": 30, "tcpFastOpen": true, "tcpNoDelay": true}' \
            /usr/local/etc/xray/config.json > /tmp/xray-fix.json && \
            mv /tmp/xray-fix.json /usr/local/etc/xray/config.json 2>/dev/null

        jq '.outbounds[0].settings.domainStrategy = "UseIPv4v6"' \
            /usr/local/etc/xray/config.json > /tmp/xray-fix.json && \
            mv /tmp/xray-fix.json /usr/local/etc/xray/config.json 2>/dev/null

        if systemctl is-active --quiet xray; then
            systemctl restart xray
            sleep 2
            systemctl is-active --quiet xray && print_success "Xray restarted" || print_error "Xray failed to restart"
        fi
        echo ""
    fi

    # Verify
    echo -e "${CYAN}Step 5: Verification...${NC}"
    echo -e "  ${YELLOW}Service binding status:${NC}"
    ss -tlnp | grep ':443' | awk '{print "    " $4 " (" $6 ")"}'
    echo ""

    print_success "Dual-stack repair complete!"
    echo ""
    read -p "Press Enter to continue..."
}

###############################################################################
# Uninstall
###############################################################################

uninstall_vpn() {
    clear
    echo -e "${RED}╔════════════════════════════════════════╗${NC}"
    echo -e "${RED}║   Uninstall VPN Services               ║${NC}"
    echo -e "${RED}╚════════════════════════════════════════╝${NC}"
    echo ""

    print_warning "This will remove all VPN services and configurations"
    read -p "Are you sure? (yes/no): " -r CONFIRM

    if [[ "$CONFIRM" != "yes" ]]; then
        print_info "Uninstall cancelled"
        return 0
    fi

    # Stop and remove Trojan
    if systemctl is-active --quiet trojan; then
        print_info "Removing Trojan..."
        systemctl stop trojan
        systemctl disable trojan > /dev/null 2>&1
        rm -f /etc/systemd/system/trojan.service
        rm -f /usr/local/bin/trojan
        rm -rf /usr/local/etc/trojan
        print_success "Trojan removed"
    fi

    for service in /etc/systemd/system/trojan-*.service; do
        if [[ -f "$service" ]]; then
            SERVICE_NAME=$(basename "$service" .service)
            systemctl stop "$SERVICE_NAME" 2>/dev/null
            systemctl disable "$SERVICE_NAME" > /dev/null 2>&1
            rm -f "$service"
            print_success "$SERVICE_NAME removed"
        fi
    done

    # Stop and remove Xray
    if systemctl is-active --quiet xray; then
        print_info "Removing Xray..."
        systemctl stop xray
        systemctl disable xray > /dev/null 2>&1
        bash -c "$(curl -L https://github.com/XTLS/Xray-install/raw/main/install-release.sh)" @ remove > /dev/null 2>&1
        rm -f /etc/systemd/system/xray.service
        print_success "Xray removed"
    fi

    for service in /etc/systemd/system/xray-*.service; do
        if [[ -f "$service" ]]; then
            SERVICE_NAME=$(basename "$service" .service)
            systemctl stop "$SERVICE_NAME" 2>/dev/null
            systemctl disable "$SERVICE_NAME" > /dev/null 2>&1
            rm -f "$service"
            print_success "$SERVICE_NAME removed"
        fi
    done

    # Stop and remove VLESS-WS
    if systemctl is-active --quiet vless-ws; then
        print_info "Removing VLESS-WS..."
        systemctl stop vless-ws
        systemctl disable vless-ws > /dev/null 2>&1
        rm -f /etc/systemd/system/vless-ws.service
        rm -f /usr/local/etc/xray/config-vless-ws.json
        rm -f /etc/nginx/sites-enabled/vless-ws
        rm -f /etc/nginx/sites-available/vless-ws
        systemctl restart nginx 2>/dev/null
        print_success "VLESS-WS removed"
    fi

    # Stop and remove WebUI
    if systemctl is-active --quiet vpn-webui; then
        print_info "Removing Web UI..."
        systemctl stop vpn-webui
        systemctl disable vpn-webui > /dev/null 2>&1
        rm -f /etc/systemd/system/vpn-webui.service
        rm -rf "$WEBUI_DIR"
        print_success "Web UI removed"
    fi

    # Remove client configs
    if [[ -d "$CLIENT_DIR" ]]; then
        rm -rf "$CLIENT_DIR"
        print_success "Client configurations removed"
    fi

    # Remove config dir
    rm -rf "$CONFIG_DIR" 2>/dev/null

    systemctl daemon-reload

    echo ""
    print_success "Uninstall complete!"
    echo ""
    read -p "Press Enter to continue..."
}

switch_protocol() {
    clear
    echo -e "${GREEN}╔════════════════════════════════════════════════╗${NC}"
    echo -e "${GREEN}║   Switch Protocol (Main IP) - Universal       ║${NC}"
    echo -e "${GREEN}╚════════════════════════════════════════════════╝${NC}"
    echo ""

    echo -e "${CYAN}Current Status:${NC}"
    for svc in trojan xray vless-ws; do
        if systemctl is-active --quiet "$svc"; then
            echo -e "  ${GREEN}[✓]${NC} $svc: Running"
        else
            echo -e "  ${RED}[✗]${NC} $svc: Stopped"
        fi
    done
    echo ""

    echo -e "${YELLOW}Options:${NC}"
    echo -e "  ${CYAN}1.${NC} Start/Restart Trojan"
    echo -e "  ${CYAN}2.${NC} Start/Restart Xray Reality"
    echo -e "  ${CYAN}3.${NC} Start/Restart VLESS-WS"
    echo -e "  ${CYAN}4.${NC} Stop all protocols"
    echo -e "  ${CYAN}5.${NC} Start all protocols"
    echo -e "  ${CYAN}0.${NC} Cancel"
    echo ""

    print_prompt "Enter your choice:"
    read -r SWITCH_CHOICE

    case $SWITCH_CHOICE in
        1)
            if [[ -f /usr/local/bin/trojan ]]; then
                systemctl stop xray 2>/dev/null
                systemctl restart trojan
                sleep 2
                systemctl is-active --quiet trojan && print_success "Trojan started" || print_error "Failed to start Trojan"
            else
                print_error "Trojan is not installed"
            fi
            ;;
        2)
            if [[ -f /usr/local/bin/xray ]] && [[ -f /usr/local/etc/xray/config.json ]]; then
                systemctl stop trojan 2>/dev/null
                systemctl restart xray
                sleep 2
                systemctl is-active --quiet xray && print_success "Xray started" || print_error "Failed to start Xray"
            else
                print_error "Xray Reality is not installed"
            fi
            ;;
        3)
            if [[ -f /usr/local/etc/xray/config-vless-ws.json ]]; then
                systemctl restart nginx 2>/dev/null
                systemctl restart vless-ws
                sleep 2
                systemctl is-active --quiet vless-ws && print_success "VLESS-WS started" || print_error "Failed to start VLESS-WS"
            else
                print_error "VLESS-WS is not installed"
            fi
            ;;
        4)
            for svc in trojan xray vless-ws; do
                systemctl stop "$svc" 2>/dev/null
            done
            print_success "All protocols stopped"
            ;;
        5)
            for svc in trojan xray vless-ws; do
                if [[ -f "/etc/systemd/system/$svc.service" ]]; then
                    systemctl start "$svc" 2>/dev/null
                fi
            done
            sleep 2
            print_success "All installed protocols started"
            ;;
        0) return 0 ;;
        *) print_error "Invalid choice" ;;
    esac

    echo ""
    read -p "Press Enter to continue..."
}

class Handler(http.server.BaseHTTPRequestHandler):
    def log_message(self, format, *args):
        pass  # Suppress logs

    def check_auth(self):
        cookies = self.headers.get('Cookie', '')
        for c in cookies.split(';'):
            c = c.strip()
            if c.startswith('auth='):
                return c[5:] == AUTH_TOKEN
        return False

    def do_GET(self):
        if self.path == '/logout':
            self.send_response(302)
            self.send_header('Set-Cookie', 'auth=; Path=/; Max-Age=0')
            self.send_header('Location', '/')
            self.end_headers()
            return

        if self.path == '/favicon.ico':
            self.send_response(204)
            self.end_headers()
            return

        auth_ok = self.check_auth()
        page = build_page(auth_ok)
        self.send_response(200)
        self.send_header('Content-Type', 'text/html; charset=utf-8')
        self.end_headers()
        self.wfile.write(page.encode())

    def do_POST(self):
        length = int(self.headers.get('Content-Length', 0))
        body = self.rfile.read(length).decode()
        params = urllib.parse.parse_qs(body)

        if self.path == '/login':
            token = params.get('token', [''])[0]
            if token == AUTH_TOKEN:
                self.send_response(302)
                self.send_header('Set-Cookie', f'auth={AUTH_TOKEN}; Path=/; HttpOnly; Max-Age=86400')
                self.send_header('Location', '/')
                self.end_headers()
            else:
                self.send_response(200)
                self.send_header('Content-Type', 'text/html')
                self.end_headers()
                self.wfile.write(build_page(False).replace('</form>', '<p class="err">Invalid token</p></form>').encode())
            return

        if self.path == '/action':
            token = params.get('token', [''])[0]
            if token != AUTH_TOKEN and not self.check_auth():
                self.send_response(302)
                self.send_header('Location', '/')
                self.end_headers()
                return

            service = params.get('service', [''])[0]
            action = params.get('action', [''])[0]

            if service in ['trojan', 'xray', 'vless-ws'] and action in ['start', 'stop', 'restart']:
                subprocess.run(f'systemctl {action} {service}', shell=True, timeout=15)

            self.send_response(302)
            self.send_header('Location', '/')
            self.end_headers()
            return

        self.send_response(302)
        self.send_header('Location', '/')
        self.end_headers()
